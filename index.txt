    <script>
        document.getElementById('qualifierForm').addEventListener('submit', function(e) {
            e.preventDefault(); // Evita que la página se recargue
            
            const btn = document.getElementById('submitBtn');
            const statusDiv = document.getElementById('statusMessage');
            
            // Cambiar estado del botón
            btn.innerHTML = 'Procesando respuestas...';
            btn.disabled = true;
            btn.classList.add('opacity-70', 'cursor-not-allowed');

            // 1. REEMPLAZA ESTA URL CON TU WEBHOOK DE MAKE.COM
            const WEBHOOK_URL = 'https://hook.make.com/tu-id-de-webhook-aqui';

            // Recopilar los datos del formulario
            const formData = {
                fecha: new Date().toISOString(),
                nombre: document.getElementById('nombre').value,
                email: document.getElementById('email').value,
                respuestas: {
                    dolor: document.querySelector('input[name="dolor"]:checked').value,
                    timeline: document.querySelector('input[name="timeline"]:checked').value,
                    liquidez: document.querySelector('input[name="liquidez"]:checked').value,
                    decision: document.querySelector('input[name="decision"]:checked').value
                }
            };

            // Enviar a Make.com
            fetch(WEBHOOK_URL, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify(formData)
            })
            .then(response => {
                // Éxito: Ocultar formulario y mostrar mensaje final
                document.getElementById('qualifierForm').innerHTML = `
                    <div class="text-center py-10">
                        <svg class="w-16 h-16 text-brand-emerald mx-auto mb-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M9 12l2 2 4-4m6 2a9 9 0 11-18 0 9 9 0 0118 0z"></path></svg>
                        <h2 class="text-2xl font-bold text-white mb-2">Evaluación Recibida</h2>
                        <p class="text-gray-400">Hemos procesado tus respuestas. Recibirás un correo electrónico en breve con el siguiente paso.</p>
                    </div>
                `;
            })
            .catch(error => {
                // Error en el envío
                statusDiv.classList.remove('hidden');
                statusDiv.classList.add('bg-red-900/50', 'text-red-400', 'border', 'border-red-800');
                statusDiv.innerText = 'Hubo un error de conexión. Por favor, intenta nuevamente.';
                
                btn.innerHTML = 'Intentar de nuevo';
                btn.disabled = false;
                btn.classList.remove('opacity-70', 'cursor-not-allowed');
            });
        });
    </script>
